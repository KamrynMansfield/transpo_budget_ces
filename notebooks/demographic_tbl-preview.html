print_demographic_shares_tbl <- function(df, col_names_to_use){
  samplesize <- nrow(df)
  
  selected_col_sums <- df |>
    select(any_of(col_names_to_use$variable)) |>
    colSums()
  
  nice_tbl <- data.frame(variable = names(selected_col_sums),
                        total = selected_col_sums) |>
    left_join(col_names_to_use, by = "variable") |>
    mutate(Share = paste0(round(total * 100 / samplesize,1),"%")) |>
    select(group, nice_name, Share) |>
    gt(groupname_col = "group",
       rowname_col = "nice_name") |>
    tab_style(
      style = cell_text(style = "italic"),
      locations = cells_row_groups(groups = everything())
    )
  
  return(nice_tbl)
}
